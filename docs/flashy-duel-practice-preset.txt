import { abilityIdFromLegacyName } from "../gameconfig/AbilityRegistry.js";
import { encodeSandboxLoadout } from "../loadout/BotLoadout.js";
import { createDefaultAbilityStrategyConfiguration } from "../botlogic/code/configuration/configurationFactories.js";
import { BOT_CODE_ACTIONS, BOT_CODE_SELECTABLES } from "../botlogic/code/contracts/BotLogicContracts.js";
import { normalizePracticeConfig } from "../practiceRoomStorage.js";
import { PUZZLE_OPPONENT_TEAM, PUZZLE_PLAYER_TEAM } from "../../pages/puzzles/puzzleRoster.js";

const MY_BOT = BOT_CODE_SELECTABLES.MY;
const OPPONENT = BOT_CODE_SELECTABLES.OPPONENT;

const ABILITIES = Object.freeze({
    HEAVY_SLASH: abilityIdFromLegacyName("heavy_slash"),
    GRENADE: abilityIdFromLegacyName("grenade"),
    FIREBALL: abilityIdFromLegacyName("fireball"),

    DASH: abilityIdFromLegacyName("dash"),
    LOCK_ON: abilityIdFromLegacyName("lock_on"),
    BASIC_STRIKE: abilityIdFromLegacyName("basic_strike"),

    ABILITIES.PHASE_STRIKE,

const OPPONENT_ABILITIES = Object.freeze([
    ABILITIES.FIREBALL,

    ABILITIES.FROST_RING,

    ABILITIES.SILENCE_PULSE,

const PLAYER_START = Object.freeze({ x: 360, y: 840, rotation: 45 });
const OPPONENT_START = Object.freeze({ x: 840, y: 360, rotation: 225 });

function always() {
    return { type: "always" };
}

function numberCondition(left, comparator, value, leftSelectable = MY_BOT, ability = null) {
    return {
        type: "expression",
        left,
        comparator,
        right: { type: "number", value },
        leftSelectable,
        ...(ability == null ? {} : { ability }),
    };
}

function pairCondition(left, comparator, value, selectable1 = MY_BOT, selectable2 = OPPONENT) {
    return {
        type: "expression",
        left,
        comparator,
        right: { type: "number", value },
        selectable1,
        selectable2,
    };
}

function booleanCondition(left, value, ability, leftSelectable = MY_BOT) {
    return {
        type: "expression",
        left,
        comparator: "eq",
        right: { type: "boolean", value },
        leftSelectable,
        ...(ability == null ? {} : { ability }),
    };
}

function abilityReady(ability, selectable = MY_BOT) {
    return booleanCondition("bot.selectedAbilityReady", true, ability, selectable);
}

function abilityReadyOrPreparing(ability, selectable = MY_BOT) {
    return [
        abilityReady(ability, selectable),
        { ...booleanCondition("bot.selectedAbilityPreparing", true, ability, selectable), join: "or" },
    ];
}

function abilityPreparing(ability, selectable = OPPONENT) {
    return booleanCondition("bot.selectedAbilityPreparing", true, ability, selectable);
}

function abilityCooldownAbove(ability, seconds, selectable = MY_BOT) {
    return numberCondition("bot.selectedAbilityCooldownMs", "gt", seconds, selectable, ability);
}

function entityExists(selectable) {
    return booleanCondition("selectable.exists", true, null, selectable);
}

function branch(id, conditions, actions, priority = 1) {
    return { id, branchType: "if", priority, conditions, actions, children: [] };
}

function root(id, priority, branches, name) {
    return {
        id,
        name,
        priority,
        branches,
    };
}

function code(roots) {
    return { ...createDefaultAbilityStrategyConfiguration(), roots };
}

function abilityAction(action, selectable = OPPONENT, fields = {}) {
    return { action, selectable, ...fields };
}

function walk(direction, selectable = OPPONENT) {
    return {
        action: BOT_CODE_ACTIONS.MOVE_WALK,
        movementMode: "target",
        movementDirection: direction,
        selectable,
    };
}

function face(selectable = OPPONENT) {
    return abilityAction(BOT_CODE_ACTIONS.ROTATE_TOWARD_TARGET, selectable);
}

function dash(selectable = OPPONENT, direction = 0) {
    return abilityAction(ABILITIES.DASH, selectable, {
        movementMode: "target",
        movementDirection: direction,
    });
}

function attack(ability, selectable = OPPONENT, fields = {}) {
    return abilityAction(ability, selectable, fields);
}

function practiceConfig(playerLoadout, opponentLoadout) {
    return normalizePracticeConfig({
        playerTeamSize: 1,
        opponentTeamSize: 1,
        initialElapsedMs: 0,
        bots: [
            {
                role: "PLAYER",
                teamNumber: PUZZLE_PLAYER_TEAM,
                slot: 1,
                loadout: playerLoadout,
                startX: PLAYER_START.x,
                startY: PLAYER_START.y,
                rotation: PLAYER_START.rotation,
                startHp: 150,
            },
            {
                role: "OPPONENT",
                teamNumber: PUZZLE_OPPONENT_TEAM,
                slot: 1,
                loadout: opponentLoadout,
                startX: OPPONENT_START.x,
                startY: OPPONENT_START.y,
                rotation: OPPONENT_START.rotation,
                startHp: 150,
            },
        ],
    });
}

export function buildFlashyDuelPreset() {
    const playerLoadout = encodeSandboxLoadout({ abilities: PLAYER_ABILITIES });
    const opponentLoadout = encodeSandboxLoadout({ abilities: OPPONENT_ABILITIES });
    return {
        id: "flashy-duel",
        label: "Flashy Duel",
        playerLoadout,
        opponentLoadout,
        playerCode: buildMyBotCode(),
        opponentCode: buildOpponentCode(),
        practiceConfig: practiceConfig(playerLoadout, opponentLoadout),
    };
}
